\begin{frame}{\bfseries Results: Spectral dimension controls squeezing scaling}

\vspace{0.3cm}
\centering
{\Huge \alert{$d\_s > 3$} leads to \alert{OAT-like} squeezing\\[0.2cm]
\alert{$2 < d\_s < 3$} yields \alert{critical} squeezing\\[0.2cm]
$d_s < 2$ no scalable squeezing}

\vspace{0.6cm}
\textcolor{accent}{\rule{\textwidth}{1pt}}

\vspace{0.3cm}
\begin{itemize}
\item \alert{Geometric prerequisite:} Existence of a \alert{percolating cluster} in the network
\item \alert{Disorder} shifts squeezing threshold, but does not affect scaling \alert{universality}
\item \alert{Spectral dimension} governs dynamic timescales and squeezing minimum scaling exponents
\end{itemize}
